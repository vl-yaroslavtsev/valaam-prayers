/**

 * @param url - URL для запроса
 * @param timeout - Таймаут в миллисекундах
 * @returns Promise с данными
 */
export async function getJson<T = any>(url: string, timeout = 10000): Promise<T> {
  const result = await fetchJson<T>(url, { timeout });
  return result.data;
}
